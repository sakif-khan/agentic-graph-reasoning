% ===================== 1 =====================
\begin{frame}{The problem}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      Language models answer factual questions \alert{fluently}, and just as
      fluently when they do not hold the fact.

      \vspace{3mm}
      In our own no-retrieval control, on WebQSP:
      \begin{itemize}
        \item $661$ entities asserted
        \item $179$ of them \alert{do not exist} in the knowledge graph
        \item that is $27.1\%$, stated with no hedge
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{The gap}
        \small
        Fluency and factuality are produced by the same mechanism, so the
        output gives the reader no signal about which one they are getting.
      \end{block}
      \vspace{2mm}
      {\small A system that is confidently wrong is worse than one that
      says it does not know.}
    \end{column}
  \end{columns}
  \takeaway{The failure is not rare, and it is not visible in the answer.}
\end{frame}

      \toprule
      \textbf{Approach} & \textbf{What it does} & \textbf{Where it stops} \\
      \midrule
      Parametric        & Answers from weights
                        & No external evidence at all \\
      Vector RAG        & Retrieves text once, then reasons
                        & Retrieval happens \emph{before} reasoning begins \\
      Static GraphRAG   & Retrieves a fixed neighbourhood
                        & Radius fixed before the question is understood \\
      Agentic navigation & Interleaves retrieval and reasoning
                        & Emits whatever the last generation call produces \\
      \bottomrule
    \end{tabular}
  \end{table}
  \takeaway{Nobody checks what the answer asserts \emph{before} it is
    delivered.}
\end{frame}

% ===================== 3 =====================
\begin{frame}{Research questions}
  \begin{enumerate}
    \item[\textbf{RQ1}] Does agentic navigation improve multi-hop factual
      accuracy --- and does the advantage \alert{grow with hop count}?
      \vspace{2mm}
    \item[\textbf{RQ2}] Given a system that already navigates the graph, what
      does a claim-level check against traversed triples \alert{contribute}?
      \vspace{2mm}
    \item[\textbf{RQ3}] Which components earn their cost, in accuracy and in
      \alert{tokens}?
  \end{enumerate}
  \vspace{3mm}
  {\small RQ2 is posed as \emph{what does it contribute}, not \emph{does it
    reduce hallucination}. The answer has several parts, and a yes/no framing
    would hide them.}
\end{frame}

% ===================== 4 =====================
\begin{frame}{AGR: an explicit state machine}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      Six nodes; the model is consulted only at decision points.
      \begin{itemize}
        \item \textbf{Planner} --- sub-objectives
        \item \textbf{Explorer} --- expands the frontier
        \item \textbf{Evaluator} --- objective met?
        \item \textbf{Backtracker} --- undo + ban list
        \item \textbf{Verifier} --- \alert{the contribution}
        \item \textbf{Answerer} --- emits what survived
      \end{itemize}
      Two cycles, both bounded by budgets rather than by model behaviour.
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      % Every gap here is load-bearing. An edge label carries fill=white, so a
      % gap shorter than the label hides the arrow it sits on and punches a
      % hole in whatever box or dashed border it overlaps. node distance is
      % therefore set from the label heights, the backtrack run from the width
      % of the word "backtrack", and the two left-hand loops sit at different
      % depths with their labels riding above and below their own runs.
      \begin{tikzpicture}[
        node distance=9mm, scale=0.60, transform shape,
        box/.style={draw=agrNode, thick, rounded corners=2pt,
          minimum height=6mm, fill=agrNode!10,
          text width=27mm, align=center, font=\small},
        vbox/.style={box, draw=agrSup, fill=agrSup!18},
        term/.style={draw=black!45, circle, inner sep=1.5pt,
          font=\scriptsize, fill=black!6},
        flow/.style={-{Stealth[length=2mm]}, semithick, draw=black!68},

        \node[box, below=of plan]      (expl)  {Explorer};
        \node[box, below=of expl]      (eval)  {Evaluator};
        \node[vbox, below=of eval]     (ver)   {Verifier};
        \node[box, below=of ver]       (ans)   {Answerer};
        \node[box, right=26mm of eval] (back)  {Backtracker};
        \draw[flow] (start) -- (plan);
        \draw[flow] (plan)  -- (expl);
        \draw[flow] (expl)  -- (eval);
        \draw[flow] (eval)  -- node[lbl] {answer} (ver);

          \node[draw=black!35, dashed, rounded corners, inner sep=1.2mm,

        \end{scope}

    \end{column}
  \end{columns}
\end{frame}

% ===================== 6 =====================
\begin{frame}{The Structural Verification Layer}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Before anything is emitted:
      \begin{enumerate}
        \item Draft answer is split into \alert{atomic claims}
        \item Each claim is checked against the \alert{triples actually
          traversed}
        \item Supported claims keep their supporting triples
        \item Unsupported claims trigger \alert{targeted re-exploration}, or
          are \alert{dropped}
      \end{enumerate}
      \vspace{2mm}
      The system therefore \textbf{hedges rather than asserts}, and returns
      every answer with its evidence.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Why \emph{structural}}
        \small
        The check is against the graph the agent walked --- not against the
        model's own judgement of its own output. A model asked to grade itself
        is not an independent check.
      \end{block}
      \vspace{2mm}
      \begin{block}{The honest limit}
        \small
        A claim can be \emph{true} and still be the wrong answer. Structural
        grounding cannot catch that --- see the echo attractor later.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ===================== 9 =====================
\begin{frame}{Making the comparison fair}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Five systems, \alert{one frozen backbone}, \alert{one budget}:
      \begin{itemize}
        \item No-retrieval (parametric control)
        \item Vector RAG
        \item Static GraphRAG
        \item Think-on-Graph (agentic)
        \item \textbf{AGR}
      \end{itemize}
      \vspace{2mm}
      Same model, temperature $0$, same $25$-call cap, same questions, same
      graph.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{What this buys}
        \small
        Differences are attributable to \textbf{architecture}, not to model
        capacity or to a bigger retrieval budget.
      \end{block}
      \vspace{1mm}
      \begin{block}{Why a parametric control}
        \small
        These benchmarks predate current models and may be partly memorised.
        Without the control we could not separate retrieval from recall.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ===================== 15 =====================
\begin{frame}{RQ2: so what does verification actually contribute?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{What it does \emph{not} do}
        \small
        Removing it changes accuracy by an amount we cannot detect:
        $p = 1.0$ on \emph{both} datasets. It does not earn its place on
        Hits@1 or F1.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{What it does do}
        \small
        \begin{itemize}
          \item Turns silent error into a \alert{hedge}: AGR hedges on
            $8.2\%$ of WebQSP against no-retrieval's $12.2\%$ error rate
          \item Attaches \alert{supporting triples} to every asserted claim
          \item Gives an \alert{auditable output contract}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  {\small This is a negative result reported as one. The layer's case rests on
  auditability, and the thesis says so rather than claiming an accuracy
  benefit the measurement does not support.}
\end{frame}

\begin{frame}{The result I did not expect}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      % A display line introducing the list below it, not a paragraph of
      % prose. Justifying a two-line heading stretches the first line to the
      % measure and leaves one word alone on the second; headings are set
      % ragged for the same reason the frame titles are.
      {\agrraggedright\textbf{Removing the planner \emph{improves} WebQSP:}\par}
      \begin{itemize}
        \item $+0.083$ F1, at $p = 0.006$
        \item $-31\%$ tokens
        \item $6.2 \rightarrow 4.0$ calls
      \end{itemize}
      \vspace{2mm}
      \small
      On ComplexWebQuestions it trends the other way ($-0.047$ F1,
      $p = 0.088$) --- not significant, but the opposite sign.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Why}
        \small
        WebQSP is predominantly one hop. Decomposing a one-hop question
        invents a second sub-objective that sends the frontier away from the
        answer.
      \end{block}
      \vspace{1mm}
      \begin{block}{What follows}
        \small
        Decomposition should be \textbf{gated on question structure}, not
        applied unconditionally. That is a design conclusion the measurement
        forced, not one it confirmed.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Every failure, read: the echo attractor}
  \begin{columns}[T]
    % 0.50/0.48 rather than 0.46/0.52: the census table needs only 48mm, but
    % the sentence above it is two lines of prose and could not be justified
    % on a 64mm measure without gaping word spaces.
    \begin{column}{0.50\textwidth}
      \small
      All $259$ remaining failures were read and labelled by hand, both
      datasets pooled:
      \vspace{1mm}
      \begin{tabular}{@{}lr@{}}
        \toprule
        \textbf{Category} & \textbf{n} \\
        \midrule
        Relation selection   & 65 \\
        Composite claim      & 47 \\
        Knowledge-graph gap  & 44 \\
        Decomposition error  & 38 \\
        Answer selection     & 15 \\
        \alert{Echo attractor} & \alert{13} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      The characteristic error of a graph navigator is \alert{not invention}.
      It is the \textbf{echo attractor}: a real, grounded entity one hop from
      the answer.

      \vspace{2mm}
      \begin{block}{Why it matters}
        \small
        Any grounding check \emph{passes} it --- because it is true. Verification
        cannot catch it by construction.
      \end{block}
      \vspace{1mm}
      {\small It appears across systems, so it is a property of the task, not
      of AGR. Naming it is what lets future work target it.}
    \end{column}
  \end{columns}
\end{frame}

% ===================== 19 =====================
\begin{frame}{The benchmark was wrong $57$ times}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Reading every failure also found questions where the \emph{benchmark},
      not the system, was at fault:
      \begin{itemize}
        \item $41$ excluded before the census
        \item $17$ found inside the census
        \item $1$ counted in both
        \item[$\Rightarrow$] \textbf{$57$ distinct questions}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Why report it}
        \small
        Reported defect rates for these two standard benchmarks are rare and
        usually anecdotal. This is a counted rate with a documented
        adjudication procedure and per-question provenance.
      \end{block}
    \end{column}
  \end{columns}
  \takeaway{A contribution that outlives this particular system.}
\end{frame}

\begin{frame}{Contributions, and what I would not claim}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Contributions}
      \small
      \begin{enumerate}
        \item AGR: a state machine over a constrained graph tool API
        \item The Structural Verification Layer and its output contract
        \item A controlled five-system comparison, one backbone, one budget
        \item The hop-count shape answering RQ1
        \item The \emph{echo attractor}, named
        \item A counted benchmark-defect rate ($57$)
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Limitations I state plainly}
      \small
      \begin{itemize}
        \item Verification earns \alert{no} detectable accuracy gain
        \item Zero ungrounded assertion is \alert{navigation}, not the layer
        \item ToG leads on the questions it finishes
        \item One backbone: results are architectural, not universal
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
